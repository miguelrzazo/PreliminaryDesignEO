\begin{frame}{Introducción. Objetivos del Proyecto}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Misión:} diseñar una constelación de nanosatélites capaz de monitorizar
      el CO\textsubscript{2} y O\textsubscript{2} atmosférico sobre EE.\,UU.
      continental con mayor resolución espacial que los instrumentos actuales.

      \bigskip
      \textbf{Requerimientos de la misión:}
      \begin{itemize}
        \item Bandas espectrales CO\textsubscript{2}: 1,61\,µm y 2,01\,µm;
              banda O\textsubscript{2}: 0,76\,µm
        \item MTF @ Nyquist $\geq 0{,}25$
        \item SNR $\geq 400$
        \item Cobertura semanal sobre EE.\,UU. ($\leq$ 7 días)
              con factor de nubosidad $\nicefrac{1}{6}$
        \item \textbf{Minimizar la masa} total puesta en órbita
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \resultbox{%
        GSD = 80\,m · SSO amanecer-anochecer\\
        2 satélites + 2 telescopios\\
        Masa total: \textbf{1,71\,kg/satélite}
      }
    \end{column}
  \end{columns}
  \note[item]{El proyecto no fabrica el satélite: establece un diseño preliminar verificable.}
  \note[item]{El GSD de 80 m supera en ×20 la resolución de OCO-2 (2,25 km).}
\end{frame}

\begin{frame}{GSD — Ground Sampling Distance}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \includegraphics[width=\textwidth]{media/image14.jpg}
    \end{column}
    \begin{column}{0.42\textwidth}
      \[
        \text{GSD} = \frac{p \cdot h}{f} \qquad f = \frac{p \cdot h}{\text{GSD}}
      \]
      donde $p$ es el tamaño de píxel, $h$ la altura orbital
      y $f$ la distancia focal.

      \bigskip
      \begin{block}{Elección: GSD = 80\,m}
        \begin{itemize}
          \item Balance entre resolución y tamaño de telescopio
          \item $\times$20 mejor que OCO-2 (2,25\,km)
          \item Permite atribución de emisiones a escala de ciudad
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \note[item]{El GSD determina la focal: a 520 km con p=15 µm → f = 97,5 mm.}
\end{frame}

\begin{frame}{Cálculo de MTF — Modelo}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      La MTF del sistema es el producto de los factores individuales:
      \[
        \text{MTF}_\text{sistema} = \text{MTF}_\text{difrac} \cdot
        \text{MTF}_\text{alin} \cdot \text{MTF}_\text{det} \cdot
        \text{MTF}_\text{margen}
      \]
      donde la MTF de difracción para una pupila circular con
      oscurecimiento central $R$ se calcula analíticamente
      (ecuaciones 6-10 a 6-15 de la memoria).

      \medskip
      \textbf{Parámetros de entrada:}
      \begin{itemize}
        \item $\lambda$, $D$, $p$ (tamaño de píxel)
        \item $f/D$ (número focal)
        \item $R$ (oscurecimiento), MTF\textsubscript{detector}, MTF\textsubscript{alineamiento}
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Requisito}
        MTF @ Nyquist $\geq 0{,}25$
      \end{block}
      \bigskip
      Frecuencia de Nyquist: $f_N = \dfrac{1}{2p}$ cy/m

      Frecuencia de corte óptico: $f_c = \dfrac{D}{\lambda}$ cy/m

      \medskip
      La MTF se evalúa en la frecuencia de Nyquist normalizada
      $u = f_N / f_c$.
    \end{column}
  \end{columns}
  \note[item]{La MTF cuantifica la degradación del contraste espacial — un valor < 0,25 implica imágenes borrosas.}
\end{frame}

% SLIDE 14 — SNR (TEORÍA)

\begin{frame}{Cálculo de SNR — Modelo}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \[
        \text{SNR} = \frac{N_e}{\sqrt{N_e + N_\text{ruido}^2}}
      \]
      donde $N_e$ (electrones de señal):
      \[
        N_e = \frac{E \cdot A_\text{pix} \cdot \eta \cdot \lambda \cdot t_\text{int}}{hc}
      \]
      y la irradiancia en el detector:
      \[
        E = \frac{\pi \tau \Delta\lambda L_\text{ref}}{1 + 4(f/D)^2}
      \]

      \textbf{Ruidos del sistema} ($N_\text{ruido}$):
      \begin{itemize}
        \item Térmico (dark), lectura, jitter, EMC, cuantización
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Requisito}
        SNR $\geq 400$
      \end{block}
      \bigskip
      \textbf{Parámetros clave:}
      \begin{itemize}
        \item $\eta$: eficiencia cuántica del detector
        \item $\tau$: transmitancia del telescopio
        \item $t_\text{int} = \text{GSD}/v_\text{orb}$
        \item $L_\text{ref} = 100\,\text{W m}^{-2}\text{sr}^{-1}\mu\text{m}^{-1}$
      \end{itemize}
    \end{column}
  \end{columns}
  \note[item]{El tiempo de integración es muy corto (~10 ms a 520 km) — por eso el SNR es el factor limitante.}
\end{frame}

\begin{frame}{Tiempo de revisita — Metodología de cobertura}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      El modelo analítico emplea el \textbf{RevisitTime Toolkit}
      (Crisp \& Livadiotti, Univ. Manchester, 2017) para calcular
      el tiempo hasta cubrir el 100\,\% de EE.\,UU. continental.

      \medskip
      \textbf{Parámetros de entrada:}
      \begin{itemize}
        \item $h$: altura orbital
        \item SW: swath (km) = $N_\text{pix} \cdot \text{GSD}$
        \item $N_\text{sat}$, $N_\text{tel}$: configuración
        \item Factor de nubosidad: $\frac{1}{6}$
        \item Factor de solapamiento: 5\,\%
      \end{itemize}

      \medskip
      Un modelo numérico de propagación de traza terrestre
      valida estos resultados simulando la cobertura sobre
      una malla de EE.\,UU.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Requisito}
        Cobertura completa $\leq 7$ días
      \end{block}

      \bigskip

\begin{frame}{Cálculo de la Masa Seca — Método de escalado}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Método de escalado por similitud geométrica
      (SMAD, Wertz 1999, Cap. 9.5):
      \[
        R = \frac{A_i}{A_o}; \quad
        W_i = K \cdot R^3 \cdot W_o; \quad
        P_i = K \cdot R^3 \cdot P_o
      \]
      donde $K = 1$ si $R \geq 0{,}5$, $K = 2$ en caso contrario.

      \medskip
      \textbf{Misiones de referencia:}
      \begin{itemize}
        \item Thematic Mapper: 0,4\,m pupila · 240\,kg
        \item SEOSAT: 0,25\,m pupila · 100\,kg
      \end{itemize}
      Se interpola entre ambas.

      \medskip
      La masa del instrumento representa el \textbf{25\,\%}
      de la masa seca total del satélite.
      Un telescopio adicional añade un \textbf{50\,\%} a la masa del instrumento.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Entrada}
        Relación $h \mapsto D_\text{min}(h)$\\
        del cruce de datos
      \end{block}
      \bigskip
      \begin{block}{Salida}
        $m_\text{seca}(h)$ para cada\\
        combinación de configuración
      \end{block}
    \end{column}
  \end{columns}
  \note[item]{El escalado geométrico es un método estándar en diseño preliminar cuando no hay datos de masa precisos.}
\end{frame}

% SLIDE 23 — MASA TOTAL (TEORÍA)
% ------------------------------------------------------------------
\begin{frame}{Cálculo de la Masa Total — Modelo de decaimiento}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Modelo de arrastre atmosférico} (USSA76):
      \[
        \frac{\mathrm{d}a}{\mathrm{d}t} = -\frac{C_d A \rho(h)\sqrt{\mu a}}{m}
      \]
      Resuelto numéricamente con \texttt{ode45} durante 8 años.
      Cuando $h < 0{,}98\,h_0$ se realiza un \textbf{impulso de Hohmann}:
      \[
        \Delta v = \sqrt{\frac{\mu}{r_1}}\!\left[\sqrt{\frac{2r_2}{r_1+r_2}}-1\right]
                 + \sqrt{\frac{\mu}{r_2}}\!\left[1-\sqrt{\frac{2r_1}{r_1+r_2}}\right]
      \]

      Masa de propelente (Tsiolkovsky, $I_{sp}=209\,\text{s}$):
      \[
        m_\text{prop} = 1{,}1 \cdot m_\text{seca}
        \left(\mathrm{e}^{\Delta v_\text{tot}/(I_{sp}g_0)}-1\right)
      \]
    \end{column}
    \begin{column}{0.42\textwidth}
      \includegraphics[width=\textwidth]{media/image26.jpg}
      {\scriptsize Transferencia de Hohmann para reimpulso orbital.}
    \end{column}
  \end{columns}
  \note[item]{El modelo numérico de decaimiento calcula el ΔV acumulado en 8 años para cada altitud.}
\end{frame}
